把产品无关的共性问题，沿 SPU 维度切成可跟踪的问题条目。零 LLM 调用——证据本来就带产品归属。

① 问题条目 共性问题 × SPU，该 SPU 证据 ≥2 条才成条目 voc_spu_issue 迁移注释记录 161 条 / 51 张有条目卡；本轮未连库复核
② SPU 容器 物化存储消息数、正负证据计数、均星、首末发布周等；正负比由看板查询派生，当前视图不存标签分布 voc_spu实测 328 张卡，其中 51 张挂有问题条目
③ 共性度 N_eff = 1 / Σ(pᵢ²) → 打 单品 / 多品 / 品线级 标签 实测品线级问题头部占比均在 14–29%，扩散是真实的
-- 与 sql/009_spu_layer.sql 对齐的关键定义
-- spu 是数组（一条评论可挂多个产品），必须 unnest 成标量再分组，
-- 否则 GROUP BY 会按整个数组值分组，产出 '{A,B}' 这种假 SPU 键。
CREATE MATERIALIZED VIEW IF NOT EXISTS voc_spu_issue AS
WITH attached AS (
SELECT DISTINCT NULLIF(btrim(s.spu), '') AS spu,
       oe.opp_id, oe.message_id, oe.seq, oe.attach_week, e.tax_path
  FROM voc_opp_evidence oe
  JOIN voc_message m USING (message_id)
  JOIN voc_evidence e
    ON e.message_id = oe.message_id AND e.seq = oe.seq
  CROSS JOIN LATERAL unnest(m.spu) AS s(spu)
 WHERE m.src_line = '电商'
   AND cardinality(m.spu) > 0
   AND NULLIF(btrim(s.spu), '') IS NOT NULL
)
SELECT spu, opp_id,
       count(*) AS evi_count,
       mode() WITHIN GROUP (ORDER BY tax_path)
         FILTER (WHERE tax_path IS NOT NULL) AS tax_path,
       min(attach_week) AS first_attach_week,
       max(attach_week) AS last_attach_week
  FROM attached
 GROUP BY spu, opp_id
HAVING count(*) >= 2;
